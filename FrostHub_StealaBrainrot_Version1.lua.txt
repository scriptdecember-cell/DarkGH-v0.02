local function MakeToggle(label, callback)
	local btn = Instance.new("TextButton")
	btn.BackgroundColor3 = OFF_BG
	btn.BorderSizePixel = 0
	btn.Text = label .. "  ●  OFF"
	btn.TextColor3 = OFF_TXT
	btn.Font = Enum.Font.GothamBold
	btn.TextSize = 12
	btn.AutoButtonColor = false
	btn.Parent = Scroll
	Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 8)
	local stroke = Instance.new("UIStroke")
	stroke.Color = OFF_STR; stroke.Thickness = 1; stroke.Parent = btn

	toggles[label] = false

	btn.MouseButton1Click:Connect(function()
		toggles[label] = not toggles[label]
		local on = toggles[label]
		btn.BackgroundColor3 = on and ON_BG or OFF_BG
		btn.TextColor3 = on and ON_TXT or OFF_TXT
		btn.Text = label .. (on and "  ●  ON" or "  ●  OFF")
		stroke.Color = on and ON_STR or OFF_STR
		callback(on)
	end)

	btn.MouseEnter:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.1), {
			BackgroundColor3 = toggles[label] and Color3.fromRGB(45, 85, 225) or Color3.fromRGB(32, 32, 52)
		}):Play()
	end)
	btn.MouseLeave:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.1), {
			BackgroundColor3 = toggles[label] and ON_BG or OFF_BG
		}):Play()
	end)
	return btn
end

local function getBestBrainrot()
	local best, bestVal = nil, -1
	local char = getChar()
	for _, obj in pairs(Workspace:GetDescendants()) do
		if obj:IsA("Model") and not obj:FindFirstChildOfClass("Humanoid") and not isPlayerChar(obj) and obj ~= char then
			local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
			if part then
				local val = obj:GetAttribute("Value") or obj:GetAttribute("Worth") or obj:GetAttribute("CashValue") or 0
				for _, v in pairs(obj:GetDescendants()) do
					if (v:IsA("NumberValue") or v:IsA("IntValue")) then
						local n = v.Name:lower()
						if n:find("value") or n:find("worth") or n:find("cash") or n:find("income") or n:find("price") then
							val = math.max(val, v.Value); break
						end
					end
				end
				if val > bestVal then best = part; bestVal = val end
			end
		end
	end
	if not best then
		local root = getRoot()
		if root then
			local dist = math.huge
			for _, obj in pairs(Workspace:GetDescendants()) do
				if obj:IsA("BasePart") then
					local n = obj.Name:lower()
					local pn = (obj.Parent and obj.Parent.Name or ""):lower()
					if n:find("brainrot") or pn:find("brainrot") or n:find("pizzanini") or n:find("burrito") then
						local d = (obj.Position - root.Position).Magnitude
						if d < dist then best = obj; dist = d end
					end
				end
			end
		end
	end
	return best
end

local function getNearestBrainrot()
	local root = getRoot()
	if not root then return nil end
	local char = getChar()
	local nearest, dist = nil, math.huge
	for _, obj in pairs(Workspace:GetDescendants()) do
		if obj:IsA("BasePart") and (not char or not obj:IsDescendantOf(char)) then
			local n = obj.Name:lower()
			local pn = (obj.Parent and obj.Parent.Name or ""):lower()
			if n:find("brainrot") or pn:find("brainrot") or n:find("pet") or pn:find("pet") or n:find("pizzanini") then
				local d = (obj.Position - root.Position).Magnitude
				if d < dist then nearest = obj; dist = d end
			end
		end
	end
	return nearest
end

local function findPlayerBase(pl)
	local pname = pl.Name:lower()
	for _, obj in pairs(Workspace:GetDescendants()) do
		local n = obj.Name:lower()
		if n:find(pname) and (n:find("base") or n:find("plot") or n:find("island") or n:find("zone") or n:find("house")) then
			return obj
		end
	end
	for _, obj in pairs(Workspace:GetChildren()) do
		if obj:IsA("Model") then
			local n = obj.Name:lower()
			if n:find("base") or n:find("plot") or n:find("island") then
				local owner = obj:FindFirstChild("Owner") or obj:FindFirstChild("PlayerName") or obj:FindFirstChild("Player")
				if owner then
					local ov = (owner:IsA("StringValue") and owner.Value)
						or (owner:IsA("ObjectValue") and owner.Value and owner.Value.Name) or ""
					if ov:lower() == pname then return obj end
				end
			end
		end
	end
	return nil
end

local function getNearestPlayer()
	local root = getRoot()
	if not root then return nil end
	local nearest, dist = nil, math.huge
	for _, pl in pairs(Players:GetPlayers()) do
		if pl ~= LP and pl.Character then
			local oRoot = pl.Character:FindFirstChild("HumanoidRootPart")
			if oRoot then
				local d = (root.Position - oRoot.Position).Magnitude
				if d < dist then nearest = pl; dist = d end
			end
		end
	end
	return nearest
end

MakeToggle("FLY V2", function(on)
	local root = getRoot()
	if not root then return end
	if on then
		local hum = getHum()
		if hum then hum.PlatformStand = true end
		flyBV = Instance.new("BodyVelocity")
		flyBV.Velocity = Vector3.zero
		flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		flyBV.Parent = root
		flyBG = Instance.new("BodyGyro")
		flyBG.P = 9e4
		flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
		flyBG.CFrame = root.CFrame
		flyBG.Parent = root
		RunService:BindToRenderStep("FrostFly", Enum.RenderPriority.Camera.Value + 1, function()
			if not toggles["FLY V2"] then RunService:UnbindFromRenderStep("FrostFly"); return end
			local cam = Workspace.CurrentCamera
			local move = Vector3.zero
			local spd = toggles["SPEED"] and 120 or 60
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then move += cam.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then move -= cam.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then move -= cam.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then move += cam.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move += Vector3.new(0, 1, 0) end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move -= Vector3.new(0, 1, 0) end
			if flyBV and flyBV.Parent then flyBV.Velocity = move.Magnitude > 0 and move.Unit * spd or Vector3.zero end
			if flyBG and flyBG.Parent then flyBG.CFrame = cam.CFrame end
		end)
	else
		RunService:UnbindFromRenderStep("FrostFly")
		if flyBV then flyBV:Destroy(); flyBV = nil end
		if flyBG then flyBG:Destroy(); flyBG = nil end
		local hum = getHum()
		if hum then hum.PlatformStand = false end
	end
end)
